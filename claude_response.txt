```python
    ("answer_candidate_idx", "Answer", "candidate_id"),
    ("answer_round_idx", "Answer", "round_id"),
]


def init_neo4j_schema() -> int:
    applied = 0
    for name, label, prop in NEO4J_CONSTRAINTS:
        try:
            run_cypher(f"CREATE CONSTRAINT {name} IF NOT EXISTS FOR (n:{label}) REQUIRE n.{prop} IS UNIQUE")
            applied += 1
        except Exception as exc:
            logger.warning("Constraint %s skipped: %s", name, exc)
    for name, label, prop in NEO4J_INDEXES:
        try:
            run_cypher(f"CREATE INDEX {name} IF NOT EXISTS FOR (n:{label}) ON (n.{prop})")
            applied += 1
        except Exception as exc:
            logger.warning("Index %s skipped: %s", name, exc)
    return applied


def init_mongo_indexes() -> list[str]:

        "emotion_logs": ["candidate_id", "interview_id", "label", "timestamp", "confidence"],

        for field in fields:
            created.append(f"{coll}.{get_collection(coll).create_index([(field, 1)])}")
    return created
```

```python
import ast
import hashlib
import json
import re
import uuid
from typing import Any, Optional

TECH_VOCAB = {

}

_CATEGORY_RULES = [
    ("Programming", {"python", "java", "c++", "c", "c#", "javascript", "typescript", "go", "rust", "kotlin", "r", "scala"}),
    ("Databases", {"sql", "mysql", "postgresql", "mongodb", "neo4j", "redis", "nosql", "database", "dbms"}),
    ("Cloud & DevOps", {"aws", "azure", "gcp", "docker", "kubernetes", "git", "linux", "devops", "terraform", "ci/cd"}),
    ("ML & AI", {"machine learning", "deep learning", "nlp", "tensorflow", "pytorch", "scikit-learn", "opencv", "yolov8", "ai", "computer vision"}),
    ("Data", {"pandas", "numpy", "tableau", "power bi", "excel", "statistics", "spark", "hadoop", "etl", "analytics", "data analysis"}),
    ("Web & APIs", {"react", "node.js", "django", "flask", "fastapi", "html", "css", "rest", "api", "spring"}),
]


def slugify(text: Any) -> str:
    return re.sub(r"[^a-z0-9+#]+", "-", str(text).lower()).strip("-")


def new_id(prefix: str) -> str:
    return f"{prefix}-{uuid.uuid4().hex[:8].upper()}"


def stable_hash(*parts: Any) -> int:
    return int(hashlib.md5("|".join(map(str, parts)).encode()).hexdigest(), 16)


def clean(value: Any) -> str:
    if value is None:
        return ""
    s = str(value).strip()
    return "" if s.lower() in {"nan", "none", "null"} else s


def dedupe(items: list[str]) -> list[str]:
    seen, out = set(), []
    for it in items:
        key = it.lower()
        if key not in seen:
            seen.add(key)
            out.append(it)
    return out


def split_list(value: Any) -> list[str]:
    """Parse a list stored as JSON, a Python repr, or delimited text (| ; ,)."""
    s = clean(value)
    if not s:
        return []
    if s[0] in "[(":
        parsed = None
        try:
            parsed = json.loads(s)
        except Exception:
            try:
                parsed = ast.literal_eval(s)
            except Exception:
                parsed = None
        if isinstance(parsed, (list, tuple)):
            return dedupe([str(x).strip() for x in parsed if str(x).strip()])
        s = s.strip("[]()")
    parts = [s]
    for sep in ("|", ";", ","):
        if sep in s:
            parts = s.split(sep)
            break
    return dedupe([p.strip().strip("'\"") for p in parts if p.strip().strip("'\"")])


def parse_years(text: Any) -> Optional[float]:
    m = re.search(r"(\d+(?:\.\d+)?)\s*\+?\s*(?:years?|yrs?)", clean(text), re.I)
    return float(m.group(1)) if m else None


def categorize_skill(name: str) -> str:
    low = name.lower()
    tokens = set(re.split(r"[^a-z0-9+#./]+", low))
    for category, keys in _CATEGORY_RULES:
        for k in keys:
            if (" " in k and k in low) or (" " not in k and k in tokens):
                return category
    return "General"


def skill_rows(names: list[str]) -> list[dict]:
    return [{"id": "SK-" + slugify(n), "name": n, "category": categorize_skill(n)} for n in dedupe(names)]
```

```python
"""CSV -> graph/document transformation. Pure Python: no database access.

Derivations (documented because the CSV has no such columns):
  * Candidate.name        = "Candidate <candidate_id>"
  * experience_years      = number found in candidate_experience text ("3 years") else null
  * Skill                 = union of the candidate `skills` column
  * TESTS_SKILL           = skill names found in question text + keywords
  * DEMONSTRATES          = skill names found in answer text + keywords
  * Role                  = distinct target_role; department = most common candidate_domain
  * REQUIRED_FOR (weight) = 2 * (#questions of the role testing the skill) + (#candidates interviewed
                            for the role who hold the skill); top 10 per role
  * Answer.score          = 10 * (0.55*keyword_coverage + 0.25*min(words/80,1) + 0.20*min(skills/2,1))
  * Emotion / Feedback / dates = deterministic pseudo-random values seeded by ids (synthetic)
"""

import csv
import random
import re
from collections import Counter, defaultdict
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any

from app.config import get_settings
from app.utils.helpers import (
    TECH_VOCAB, categorize_skill, clean, parse_years, slugify, split_list, stable_hash,
)

DATASET_NAME = "synthetic_interview_dataset.csv"
EXPECTED_COLUMNS = [
    "candidate_id", "candidate_domain", "education", "skills", "candidate_experience", "job_type",
    "round_id", "question_id", "question", "question_category", "target_role", "difficulty",
    "question_experience", "answer_id", "answer", "keywords",
]
PHASES = [("start", 0), ("middle", 60), ("end", 150)]


def find_dataset() -> Path:
    here = Path(__file__).resolve()
    options = []
    if get_settings().dataset_path:
        options.append(Path(get_settings().dataset_path))
    options += [
        here.parents[2] / "data" / DATASET_NAME,
        here.parents[3] / "data" / DATASET_NAME,
        Path("/app/data") / DATASET_NAME,
    ]
    for p in options:
        if p.exists():
            return p
    raise FileNotFoundError(
        f"{DATASET_NAME} not found. Put it in <project>/data/ . Looked in: " + ", ".join(map(str, options))
    )


def read_rows(path: Path) -> list[dict[str, str]]:
    with open(path, newline="", encoding="utf-8-sig") as f:
        return [{(k or "").strip().lower(): (v or "").strip() for k, v in r.items()} for r in csv.DictReader(f)]


def _pattern(name: str):
    return re.compile(r"(?<![A-Za-z0-9+#])" + re.escape(name) + r"(?![A-Za-z0-9+#])", re.I)


def score_answer(answer: str, keywords: list[str], n_skills: int = 0):
    """Deterministic answer score. Returns (score, keyword_coverage, word_count, matched_keywords)."""
    text = (answer or "").lower()
    words = len(text.split())
    hits = [k for k in keywords if k.lower() in text]
    cov = len(hits) / len(keywords) if keywords else 0.5
    score = 10 * (0.55 * cov + 0.25 * min(words / 80, 1.0) + 0.20 * min(n_skills / 2, 1.0))
    return round(max(1.0, min(10.0, score)), 1), round(cov, 2), words, hits


def _emotion(rng: random.Random, score: float) -> tuple[str, float]:
    p = score / 10
    weights = {
        "confident": 0.5 * p + 0.05, "enthusiastic": 0.25 * p + 0.03, "neutral": 0.30,
        "hesitant": 0.35 * (1 - p) + 0.05, "nervous": 0.40 * (1 - p) + 0.05,
    }
    label = rng.choices(list(weights), weights=list(weights.values()))[0]
    return label, round(0.55 + 0.4 * rng.random(), 2)


def transform(rows: list[dict[str, str]]) -> dict[str, Any]:
    if not rows:
        raise ValueError("Dataset is empty")
    missing_cols = [c for c in EXPECTED_COLUMNS if c not in rows[0]]

    vocab: dict[str, str] = {}
    for r in rows:
        for s in split_list(r.get("skills")):
            vocab.setdefault(s.lower(), s)
    skills = {k: {"skill_id": "SK-" + slugify(v), "name": v, "category": categorize_skill(v)} for k, v in vocab.items()}
    patterns = {k: _pattern(v) for k, v in vocab.items()}

    def find_skills(text: str) -> list[str]:
        return [k for k, p in patterns.items() if p.search(text)]

    candidates: dict[str, dict] = {}
    cand_skill_keys: dict[str, list[str]] = {}
    cand_rounds: dict[str, dict[str, int]] = defaultdict(dict)
    questions: dict[str, dict] = {}
    rounds: dict[str, dict] = {}
    interviews: dict[str, dict] = {}
    answers, emotion_logs = [], []
    q_skills, a_skills, c_skills = set(), set(), set()
    tested_by_role: dict[str, Counter] = defaultdict(Counter)
    role_domains: dict[str, Counter] = defaultdict(Counter)
    role_candidates: dict[str, set] = defaultdict(set)
    cand_roles: set[tuple[str, str]] = set()
    companies: dict[str, dict] = {}
    cand_company: set[tuple[str, str]] = set()

    for i, r in enumerate(rows):
        cid = r.get("candidate_id") or r.get("candidate_source_id") or f"C{i + 1:03d}"
        qid = r.get("question_id") or f"Q{i + 1:03d}"
        aid = r.get("answer_id") or f"A{i + 1:03d}"
        role = r.get("target_role") or r.get("source_role") or "General"
        rraw = r.get("round_id") or "R1"
        round_key = f"{cid}__{rraw}"
        qtext, atext = r.get("question", ""), r.get("answer", "")
        kws = split_list(r.get("keywords"))
        diff = r.get("difficulty") or "Medium"
        cat = r.get("question_category") or "General"

        if cid not in candidates:
            keys = [s.lower() for s in split_list(r.get("skills")) if s.lower() in skills]
            cand_skill_keys[cid] = keys
            candidates[cid] = {
                "candidate_id": cid, "name": f"Candidate {cid}", "email": None,
                "experience_years": parse_years(r.get("candidate_experience")),
                "education": r.get("education") or None, "domain": r.get("candidate_domain") or None,
                "job_type": r.get("job_type") or None, "skills": [skills[k]["name"] for k in keys],
                "source_id": r.get("candidate_source_id") or None,
                "experience_text": r.get("candidate_experience") or None,
            }
            for k in keys:
                c_skills.add((cid, skills[k]["skill_id"]))

        role_id = "R-" + slugify(role)
        role_domains[role][r.get("candidate_domain") or "General"] += 1
        role_candidates[role].add(cid)
        cand_roles.add((cid, role_id))

        tested = find_skills(qtext + " " + " ".join(kws))
        shown = find_skills(atext + " " + " ".join(kws))

        if qid not in questions:
            questions[qid] = {
                "question_id": qid, "text": qtext, "difficulty": diff, "category": cat,
                "target_role": role, "role_id": role_id, "experience": r.get("question_experience") or None,
                "keywords": kws,
            }
            for k in tested:
                q_skills.add((qid, skills[k]["skill_id"]))
                tested_by_role[role][k] += 1

        if rraw not in cand_rounds[cid]:
            cand_rounds[cid][rraw] = len(cand_rounds[cid]) + 1
        rnum = cand_rounds[cid][rraw]
        start = datetime(2025, 1, 6, 10, 0) + timedelta(days=stable_hash(round_key) % 180)
        if round_key not in rounds:
            rounds[round_key] = {
                "round_id": round_key, "round_label": rraw, "round_number": rnum, "candidate_id": cid,
                "date": start.date().isoformat(), "target_role": role,
            }
            interviews[round_key] = {
                "_id": round_key, "interview_id": round_key, "candidate_id": cid, "round_id": round_key,
                "round_label": rraw, "round_number": rnum, "target_role": role, "started_at": start,
                "status": "completed", "transcript": [], "qa": [], "emotion_timeline": [],
                "metadata": {k: v for k, v in {
                    "synthetic": True, "dataset": DATASET_NAME, "candidate_domain": r.get("candidate_domain"),
                    "job_type": r.get("job_type"), "source_role": r.get("source_role"),
                    "source_type": r.get("source_type"), "answer_source": r.get("answer_source"),
                    "merge_note": r.get("merge_note"),
                }.items() if v not in (None, "")},
            }
        iv = interviews[round_key]

        score, cov, words, hits = score_answer(atext, kws, len(shown))
        rng = random.Random(stable_hash(aid))
        t = 120 + len(iv["qa"]) * 240
        samples = []
        for phase, off in PHASES:
            label, conf = _emotion(rng, score)
            samples.append({"phase": phase, "timestamp_sec": t + off, "label": label, "confidence": conf})
        dom = samples[1]
        stamp = (start + timedelta(seconds=t)).isoformat()
        missed = [k for k in kws if k not in hits]

        answers.append({
            "answer_id": aid, "candidate_id": cid, "question_id": qid, "round_id": round_key,
            "transcript_text": atext, "score": score, "keyword_coverage": cov, "word_count": words,
            "timestamp": stamp, "emotion_id": "EMO-" + aid, "emotion_label": dom["label"],
            "emotion_confidence": dom["confidence"], "feedback_id": "FB-" + aid,
            "strengths": ", ".join(hits[:4]) or "Relevant answer",
            "weaknesses": ", ".join(missed[:4]) or "No major gaps detected",
            "summary": f"Scored {score}/10 on a {diff} {cat} question (keyword coverage {cov:.0%}).",
        })
        for k in shown:
            a_skills.add((aid, skills[k]["skill_id"]))

        iv["transcript"] += [
            {"speaker": "interviewer", "text": qtext, "t": t},
            {"speaker": "candidate", "text": atext, "t": t + 30},
        ]
        iv["qa"].append({
            "question_id": qid, "question": qtext, "category": cat, "difficulty": diff, "answer_id": aid,
            "answer": atext, "score": score, "skills_tested": [skills[k]["name"] for k in tested],
            "keywords": kws, "timestamp": stamp, "timestamp_sec": t,
            "emotion": {"label": dom["label"], "confidence": dom["confidence"]},
        })
        for idx, s in enumerate(samples):
            iv["emotion_timeline"].append({**s, "question_id": qid})
            emotion_logs.append({
                "_id": f"{aid}:{idx}", "interview_id": round_key, "candidate_id": cid, "question_id": qid,
                "answer_id": aid, "label": s["label"], "confidence": s["confidence"], "phase": s["phase"],
                "timestamp_sec": s["timestamp_sec"], "timestamp": start + timedelta(seconds=s["timestamp_sec"]),
                "target_role": role, "synthetic": True,
            })

        company = r.get("company")
        if company:
            companies["CO-" + slugify(company)] = {
                "company_id": "CO-" + slugify(company), "name": company, "industry": r.get("industry") or None,
            }
            cand_company.add((cid, "CO-" + slugify(company)))

    # Roles + REQUIRED_FOR
    roles, role_skills = {}, []
    for role, doms in role_domains.items():
        rid = "R-" + slugify(role)
        roles[rid] = {"role_id": rid, "title": role, "department": doms.most_common(1)[0][0]}
        holders = Counter(k for c in role_candidates[role] for k in cand_skill_keys[c])
        weights = {k: tested_by_role[role][k] * 2 + holders[k] for k in set(tested_by_role[role]) | set(holders)}
        for k, w in sorted(weights.items(), key=lambda kv: (-kv[1], kv[0]))[:10]:
            if w > 0:
                role_skills.append({"skill_id": skills[k]["skill_id"], "role_id": rid, "weight": w})

    techs = {k: {"tech_id": "T-" + slugify(v["name"]), "name": v["name"]} for k, v in skills.items() if k in TECH_VOCAB}
    skill_tech = [{"skill_id": skills[k]["skill_id"], "tech_id": t["tech_id"]} for k, t in techs.items()]

    pair = lambda s, a, b: [{a: x, b: y} for x, y in sorted(s)]
    data = {
        "candidates": list(candidates.values()), "questions": list(questions.values()),
        "skills": [{k: v for k, v in s.items()} for s in skills.values()], "roles": list(roles.values()),
        "technologies": list(techs.values()), "companies": list(companies.values()),
        "rounds": list(rounds.values()), "answers": answers,
        "candidate_roles": [{"candidate_id": c, "role_id": r} for c, r in sorted(cand_roles)],
        "candidate_skills": pair(c_skills, "candidate_id", "skill_id"),
        "question_skills": pair(q_skills, "question_id", "skill_id"),
        "answer_skills": pair(a_skills, "answer_id", "skill_id"),
        "candidate_companies": pair(cand_company, "candidate_id", "company_id"),
        "role_skills": role_skills, "skill_tech": skill_tech,
        "mongo_interviews": list(interviews.values()), "emotion_logs": emotion_logs,
```

```python
"""Shared graph helpers used by several services."""

from typing import Any

from app.database.neo4j import run_cypher


def run(query: str, **params: Any) -> list[dict]:
    return run_cypher(query, params)


def candidate_skill_map() -> dict[str, dict]:
    rows = run("""
        MATCH (c:Candidate)
        OPTIONAL MATCH (c)-[:HAS_SKILL]->(s:Skill)
        RETURN c.candidate_id AS id, c.name AS name, c.domain AS domain, collect(s.name) AS skills
    """)
    return {r["id"]: r for r in rows}


def role_requirements() -> list[dict]:
    return run("""
        MATCH (r:Role)
        OPTIONAL MATCH (s:Skill)-[rel:REQUIRED_FOR]->(r)
        WITH r, s, rel ORDER BY rel.weight DESC
        RETURN r.role_id AS role_id, r.title AS title, r.department AS department,
               collect(CASE WHEN s IS NULL THEN NULL ELSE {name: s.name, weight: rel.weight} END) AS required
        ORDER BY title
    """)


def interviewed_roles() -> dict[str, list[str]]:
    rows = run("MATCH (c:Candidate)-[:INTERVIEWED_FOR]->(r:Role) RETURN c.candidate_id AS id, collect(r.title) AS roles")
    return {r["id"]: r["roles"] for r in rows}


def coverage(candidate_skills: list[str], required_names: list[str]) -> dict:
    have = {s.lower() for s in candidate_skills}
    matched = [n for n in required_names if n.lower() in have]
    missing = [n for n in required_names if n.lower() not in have]
    pct = round(100 * len(matched) / len(required_names), 1) if required_names else 0.0
    return {"matched": matched, "missing": missing, "coverage": pct}


def jaccard(a: list[str], b: list[str]) -> tuple[list[str], float]:
    sa, sb = {s.lower(): s for s in a}, {s.lower(): s for s in b}
    shared = [sa[k] for k in sa if k in sb]
    union = len(set(sa) | set(sb))
    return shared, round(100 * len(shared) / union, 1) if union else 0.0


def best_role(cand_skills: list[str], reqs: list[dict], prefer: list[str] | None = None) -> dict | None:
    pool = [r for r in reqs if prefer and r["title"] in prefer] or reqs
    best = None
    for r in pool:
        names = [x["name"] for x in r["required"]]
        cov = coverage(cand_skills, names)
        if best is None or cov["coverage"] > best["coverage"]:
            best = {"role_id": r["role_id"], "title": r["title"], "required": names, **cov}
    return best
```

```python
def list_candidates(q=None, domain=None, job_type=None) -> dict:
    rows = run("""
        MATCH (c:Candidate)
        OPTIONAL MATCH (c)-[:ANSWERED]->(a:Answer)
        WITH c, avg(a.score) AS avg_score, count(a) AS answers
        RETURN c.candidate_id AS candidate_id, c.name AS name, c.domain AS domain,
               c.education AS education, c.experience_years AS experience_years,
               c.job_type AS job_type, avg_score, answers
        ORDER BY candidate_id
    """)
    smap, reqs, roles = candidate_skill_map(), role_requirements(), interviewed_roles()
    items = []
    for r in rows:
        skills = smap.get(r["candidate_id"], {}).get("skills", [])
        primary = best_role(skills, reqs, roles.get(r["candidate_id"]))
        overall = best_role(skills, reqs)
        items.append({
            **r, "skills": skills,
            "avg_score": round(r["avg_score"], 2) if r["avg_score"] is not None else None,
            "primary_role": primary["title"] if primary else None,
            "skill_coverage": primary["coverage"] if primary else 0,
            "best_role": overall["title"] if overall else None,
            "best_role_compat": overall["coverage"] if overall else 0,
        })
    facets = {
        "domains": sorted({i["domain"] for i in items if i["domain"]}),
        "job_types": sorted({i["job_type"] for i in items if i["job_type"]}),
    }
    if q:
        ql = q.lower()
        items = [i for i in items if ql in " ".join(
            [str(i["candidate_id"]), str(i["name"]), str(i["education"] or ""), " ".join(i["skills"])]).lower()]
    if domain:
        items = [i for i in items if i["domain"] == domain]
    if job_type:
        items = [i for i in items if i["job_type"] == job_type]
    return {"items": items, "total": len(items), "facets": facets}
```

```python
                ids=[x.question_id for x in p.qa])

        q = qmap[item.question_id]
```

```python
"""Deterministic, explainable candidate evaluation (no LLM, no paid API)."""

from app.services.shared import best_role, candidate_skill_map, interviewed_roles, role_requirements, run
from app.utils.errors import NotFoundError

LABEL = "System-generated evaluation based on synthetic interview data."
DIFF_WEIGHT = {"easy": 1.0, "medium": 1.5, "hard": 2.0}
BEHAVIORAL_HINTS = ("behav", "hr", "situation", "soft", "culture", "leadership", "communication")


def _is_behavioral(category: str | None) -> bool:
    c = (category or "").lower()
    return any(h in c for h in BEHAVIORAL_HINTS)


def _avg(values: list[float]) -> float:
    return sum(values) / len(values) if values else 0.0


def compute_evaluation(answers: list[dict], coverage_pct: float, matched: list[str], missing: list[str],
                       role: str | None = None) -> dict:
    """Pure function (unit-testable). Scores are on a 0-10 scale; coverage is a percentage."""
    scored = [a for a in answers if a.get("score") is not None]
    tech_pool = [a for a in scored if not _is_behavioral(a.get("category"))] or scored
    beh_pool = [a for a in scored if _is_behavioral(a.get("category"))]

    wsum = sum(DIFF_WEIGHT.get((a.get("difficulty") or "").lower(), 1.0) for a in tech_pool)
    technical = sum(a["score"] * DIFF_WEIGHT.get((a.get("difficulty") or "").lower(), 1.0) for a in tech_pool) / wsum if wsum else 0.0
    length_score = _avg([min((a.get("word_count") or 0) / 80, 1.0) * 10 for a in scored])
    beh_score = _avg([a["score"] for a in beh_pool]) if beh_pool else _avg([a["score"] for a in scored])
    communication = 0.5 * length_score + 0.5 * beh_score
    overall = 0.5 * technical + 0.2 * communication + 0.3 * (coverage_pct / 10)

    by_cat: dict[str, list[float]] = {}
    by_skill: dict[str, list[float]] = {}
    for a in scored:
        by_cat.setdefault(a.get("category") or "General", []).append(a["score"])
        for s in a.get("skills") or []:
            by_skill.setdefault(s, []).append(a["score"])

    strengths = [f"Strong performance in {c} questions (average {_avg(v):.1f}/10)" for c, v in by_cat.items() if _avg(v) >= 7]
    strengths += [f"Skill evidence: {s} (average {_avg(v):.1f}/10)" for s, v in
                  sorted(by_skill.items(), key=lambda kv: -_avg(kv[1])) if _avg(v) >= 7][:3]
    if matched:
        strengths.append("Holds role skills: " + ", ".join(matched[:5]))
    weaknesses = [f"Below-average performance in {c} questions (average {_avg(v):.1f}/10)" for c, v in by_cat.items() if _avg(v) < 5]
    weaknesses += [f"Missing role skill: {m}" for m in missing[:5]]
    hard_low = [a for a in scored if (a.get("difficulty") or "").lower() == "hard" and a["score"] < 5]
    if hard_low:
        weaknesses.append(f"{len(hard_low)} hard question(s) scored below 5/10")

    summary = (f"Overall {overall:.1f}/10 (technical {technical:.1f}, communication {communication:.1f}); "
               f"skill coverage {coverage_pct:.0f}%" + (f" for {role}." if role else "."))
    explanation = [
        f"Technical = difficulty-weighted mean of answer scores (Easy 1.0, Medium 1.5, Hard 2.0) over {len(tech_pool)} answers = {technical:.2f}.",
        f"Communication = 0.5 * answer-length score ({length_score:.2f}) + 0.5 * behavioural/overall score ({beh_score:.2f}) = {communication:.2f}.",
        f"Skill coverage = matched required skills / required skills for the role = {coverage_pct:.1f}%.",
        f"Overall = 0.5*technical + 0.2*communication + 0.3*(coverage/10) = {overall:.2f}.",
        "Answer score = 10 * (0.55*keyword coverage + 0.25*length factor + 0.20*skill evidence), computed at seed time.",
    ]
    return {
        "overall_score": round(overall, 1), "technical_score": round(technical, 1),
        "communication_score": round(communication, 1), "skill_coverage": round(coverage_pct, 1),
        "strengths": strengths or ["No standout strengths detected from the available data"],
        "weaknesses": weaknesses or ["No significant weaknesses detected from the available data"],
        "summary": summary, "explanation": explanation,
        "breakdown": {"answers": len(scored), "by_category": {c: round(_avg(v), 2) for c, v in by_cat.items()},
                      "matched_skills": matched, "missing_skills": missing},
    }


def evaluate_candidate(cid: str) -> dict:
    cand = candidate_skill_map().get(cid)
    if not cand:
        raise NotFoundError(f"Candidate {cid} not found")
    answers = run("""
        MATCH (:Candidate {candidate_id:$id})-[:ANSWERED]->(a:Answer)-[:IN_RESPONSE_TO]->(q:Question)
        OPTIONAL MATCH (a)-[:DEMONSTRATES]->(s:Skill)
        RETURN a.answer_id AS answer_id, a.score AS score, a.word_count AS word_count,
               q.difficulty AS difficulty, q.category AS category, collect(DISTINCT s.name) AS skills
    """, id=cid)
    role = best_role(cand["skills"], role_requirements(), interviewed_roles().get(cid))
    result = compute_evaluation(answers, role["coverage"] if role else 0.0,
                                role["matched"] if role else [], role["missing"] if role else [],
                                role["title"] if role else None)
    return {"candidate_id": cid, "name": cand["name"], "role": role["title"] if role else None,
            "label": LABEL, **result}
```

```python
from app.services.shared import candidate_skill_map, coverage, jaccard, role_requirements
from app.utils.errors import NotFoundError

LABEL = "System-generated role compatibility (skill overlap), not a hiring decision."


def _candidate(cid: str) -> dict:
    c = candidate_skill_map().get(cid)
    if not c:
        raise NotFoundError(f"Candidate {cid} not found")
    return c


def role_recommendations(cid: str) -> dict:
    cand = _candidate(cid)
    roles = []
    for r in role_requirements():
        names = [x["name"] for x in r["required"]]
        roles.append({"role_id": r["role_id"], "title": r["title"], "required": names,
                      **coverage(cand["skills"], names)})
    roles.sort(key=lambda x: -x["coverage"])
    return {"candidate_id": cid, "name": cand["name"], "label": LABEL, "roles": roles}


def skill_gap(cid: str, role_id: str) -> dict:
    cand = _candidate(cid)
    role = next((r for r in role_requirements() if r["role_id"] == role_id or r["title"] == role_id), None)
    if not role:
        raise NotFoundError(f"Role {role_id} not found")
    names = [x["name"] for x in role["required"]]
    return {"candidate_id": cid, "role": {"role_id": role["role_id"], "title": role["title"]},
            "required": names, **coverage(cand["skills"], names)}


def similar_candidates(cid: str, limit: int = 5) -> dict:
    cand = _candidate(cid)
    out = []
    for oid, o in candidate_skill_map().items():
        if oid == cid:
            continue
        shared, pct = jaccard(cand["skills"], o["skills"])
        if shared:
            out.append({"candidate_id": oid, "name": o["name"], "domain": o["domain"],
                        "shared_skills": shared, "similarity": pct})
    out.sort(key=lambda x: -x["similarity"])
    return {"candidate_id": cid, "method": "Jaccard similarity over HAS_SKILL relationships", "items": out[:limit]}
```

```python
"""Graph payloads for react-force-graph-2d, read live from Neo4j."""

from app.services.shared import run
from app.utils.errors import NotFoundError

DEFAULT_TYPES = ["Candidate", "Question", "Skill", "Role", "Company", "Answer", "Technology", "InterviewRound"]
_KEY = ("labels({v})[0] + ':' + toString(coalesce({v}.candidate_id, {v}.question_id, {v}.skill_id, {v}.role_id, "
        "{v}.company_id, {v}.answer_id, {v}.tech_id, {v}.round_id, {v}.emotion_id, {v}.feedback_id))")
_NAME = "coalesce({v}.name, {v}.title, {v}.answer_id, {v}.round_id, {v}.text)"


def _k(v: str) -> str:
    return _KEY.replace("{v}", v)


def _shorten(value, n=300):
    return value[:n] + "…" if isinstance(value, str) and len(value) > n else value


def _node(r: dict) -> dict:
    name = str(r["name"] or r["id"])
    return {"id": r["id"], "type": r["type"], "name": name if len(name) <= 60 else name[:57] + "…",
            "props": {k: _shorten(v) for k, v in (r["props"] or {}).items()}}


def _stats(nodes: list, links: list) -> dict:
    by_type: dict[str, int] = {}
    for n in nodes:
        by_type[n["type"]] = by_type.get(n["type"], 0) + 1
    return {"nodes": len(nodes), "links": len(links), "by_type": by_type}


def _links_between(keys: list[str]) -> list[dict]:
    rows = run(f"""
        MATCH (a)-[r]->(b)
        WHERE {_k('a')} IN $keys AND {_k('b')} IN $keys
        RETURN {_k('a')} AS source, {_k('b')} AS target, type(r) AS type
    """, keys=keys)
    return rows


def full_graph(types: list[str] | None = None) -> dict:
    types = [t for t in (types or DEFAULT_TYPES) if t in DEFAULT_TYPES + ["Emotion", "Feedback"]]
    nodes = [_node(r) for r in run(f"""
        MATCH (n) WHERE labels(n)[0] IN $types
        RETURN labels(n)[0] AS type, {_k('n')} AS id, {_NAME.replace('{v}', 'n')} AS name, properties(n) AS props
    """, types=types)]
    links = run(f"""
        MATCH (a)-[r]->(b)
        WHERE labels(a)[0] IN $types AND labels(b)[0] IN $types
        RETURN {_k('a')} AS source, {_k('b')} AS target, type(r) AS type
    """, types=types)
    return {"nodes": nodes, "links": links, "stats": _stats(nodes, links)}


def candidate_graph(cid: str) -> dict:
    rows = run(f"""
        MATCH (c:Candidate {{candidate_id:$id}})
        CALL {{
            WITH c RETURN c AS n
            UNION WITH c MATCH (c)-[:ANSWERED]->(n:Answer) RETURN n
            UNION WITH c MATCH (c)-[:ANSWERED]->(:Answer)-[:IN_RESPONSE_TO]->(n:Question) RETURN n
            UNION WITH c MATCH (c)-[:PARTICIPATES_IN]->(n:InterviewRound) RETURN n
            UNION WITH c MATCH (c)-[:HAS_SKILL]->(n:Skill) RETURN n
            UNION WITH c MATCH (c)-[:INTERVIEWED_FOR]->(n:Role) RETURN n
            UNION WITH c MATCH (c)-[:HAS_SKILL]->(:Skill)-[:RELATED_TO]->(n:Technology) RETURN n
        }}
        RETURN DISTINCT labels(n)[0] AS type, {_k('n')} AS id, {_NAME.replace('{v}', 'n')} AS name, properties(n) AS props
    """, id=cid)
    if not rows:
        raise NotFoundError(f"Candidate {cid} not found")
    nodes = [_node(r) for r in rows]
    links = _links_between([n["id"] for n in nodes])
    return {"nodes": nodes, "links": links, "stats": _stats(nodes, links)}
```

```python
def _with_projection(name: str, nodes: list[str], rels: dict, algo: str) -> list[dict]:
    try:
        run("CALL gds.graph.drop($name, false)", name=name)
        run("CALL gds.graph.project($name, $nodes, $rels)", name=name, nodes=nodes, rels=rels)
        return run(algo, name=name)
    finally:
        try:
            run("CALL gds.graph.drop($name, false)", name=name)
        except Exception:
            pass

        return {"engine": "gds", "algorithm": "Node Similarity", "items": items}
    except Exception as exc:
        smap = candidate_skill_map()
        ids, pairs = sorted(smap), []
        for i, a in enumerate(ids):
            for b in ids[i + 1:]:
                shared, pct = jaccard(smap[a]["skills"], smap[b]["skills"])
                if shared:
                    pairs.append({"a": a, "b": b, "similarity": round(pct / 100, 3)})
        pairs.sort(key=lambda x: -x["similarity"])
        return {"engine": "fallback", "algorithm": "Jaccard (Python)", "note": f"GDS unavailable: {str(exc)[:120]}", "items": pairs[:10]}


def gds_analytics() -> dict:
    return {"pagerank": _pagerank(), "louvain": _louvain(), "similarity": _similarity()}

    emo, iv = get_collection("emotion_logs"), get_collection("raw_interviews")
    avg_conf = list(emo.aggregate([{"$group": {"_id": None, "avg": {"$avg": "$confidence"}, "samples": {"$sum": 1}}},
                                   {"$project": {"_id": 0, "avg": {"$round": ["$avg", 3]}, "samples": 1}}]))
    return {

        "avg_answer_score": (list(iv.aggregate([
            {"$unwind": "$qa"}, {"$group": {"_id": None, "avg": {"$avg": "$qa.score"}, "answers": {"$sum": 1}}},
```

```python
    settings = get_settings()

        return {"connected": False, "database": settings.mongodb_database}
    db = get_mongo_database()

        sample = db[name].find_one({}, {"transcript": 0})

            sample["qa"] = sample["qa"][:1]
        if sample and isinstance(sample.get("emotion_timeline"), list):
            sample["emotion_timeline"] = sample["emotion_timeline"][:2]
        cols.append({"name": name, "documents": db[name].count_documents({}), "sample": sample})

    out = {"neo4j": None, "mongodb": {}}
    try:
        out["neo4j"] = _neo4j_indexes()
    except Exception:
        out["neo4j"] = {"constraints": [], "indexes": [], "error": "Neo4j unavailable"}
    try:
        db = get_mongo_database()
        for name in sorted(db.list_collection_names()):
            out["mongodb"][name] = [{"name": k, "keys": [list(x) for x in v["key"]]} for k, v in db[name].index_information().items()]
    except Exception:
        out["mongodb"] = {"error": "MongoDB unavailable"}
```

```python
            rd.candidate_id=r.candidate_id, rd.target_role=r.target_role

        MATCH (ro:Role {role_id:r.role_id}) MERGE (c)-[:INTERVIEWED_FOR]->(ro)""",
    "answers": """UNWIND $rows AS r
        MATCH (c:Candidate {candidate_id:r.candidate_id}) MATCH (q:Question {question_id:r.question_id})
        MATCH (rd:InterviewRound {round_id:r.round_id})
        MERGE (a:Answer {answer_id:r.answer_id})
        SET a.transcript_text=r.transcript_text, a.score=r.score, a.keyword_coverage=r.keyword_coverage,
            a.word_count=r.word_count, a.timestamp=r.timestamp, a.candidate_id=r.candidate_id,
            a.round_id=r.round_id, a.question_id=r.question_id
        MERGE (c)-[:ANSWERED]->(a) MERGE (a)-[:IN_RESPONSE_TO]->(q) MERGE (rd)-[:INCLUDES]->(q)
        MERGE (f:Feedback {feedback_id:r.feedback_id})
        SET f.strengths=r.strengths, f.weaknesses=r.weaknesses, f.summary=r.summary
        MERGE (a)-[:RECEIVED]->(f)
        MERGE (e:Emotion {emotion_id:r.emotion_id})

    "answer_skills": "UNWIND $rows AS r MATCH (a:Answer {answer_id:r.answer_id}) MATCH (s:Skill {skill_id:r.skill_id}) MERGE (a)-[:DEMONSTRATES]->(s)",
    "role_skills": "UNWIND $rows AS r MATCH (s:Skill {skill_id:r.skill_id}) MATCH (x:Role {role_id:r.role_id}) MERGE (s)-[rel:REQUIRED_FOR]->(x) SET rel.weight=r.weight",
    "skill_tech": "UNWIND $rows AS r MATCH (s:Skill {skill_id:r.skill_id}) MATCH (t:Technology {tech_id:r.tech_id}) MERGE (s)-[:RELATED_TO]->(t)",
    "candidate_companies": "UNWIND $rows AS r MATCH (c:Candidate {candidate_id:r.candidate_id}) MATCH (co:Company {company_id:r.company_id}) MERGE (c)-[:APPLIED_TO]->(co)",
}
ORDER = [("skills", "skills"), ("roles", "roles"), ("technologies", "techs"), ("companies", "companies"),
         ("candidates", "candidates"), ("questions", "questions"), ("rounds", "rounds"),
         ("candidate_roles", "candidate_roles"), ("answers", "answers"), ("candidate_skills", "candidate_skills"),
         ("question_skills", "question_skills"), ("answer_skills", "answer_skills"), ("role_skills", "role_skills"),
         ("skill_tech", "skill_tech"), ("candidate_companies", "candidate_companies")]


def _batches(rows: list, size: int = 500):
    for i in range(0, len(rows), size):
        yield rows[i:i + size]


def load_neo4j(data: dict) -> None:
    for key, qname in ORDER:
        rows = data.get(key) or []
        for batch in _batches(rows):
            run_cypher(Q[qname], {"rows": batch})


def load_mongo(data: dict) -> None:
    iv, emo = get_collection("raw_interviews"), get_collection("emotion_logs")
    for coll, docs in ((iv, data["mongo_interviews"]), (emo, data["emotion_logs"])):
        for batch in _batches(docs, 500):
            coll.bulk_write([ReplaceOne({"_id": d["_id"]}, d, upsert=True) for d in batch], ordered=False)


def stats() -> dict:
    labels = {r["l"]: r["c"] for r in run_cypher("MATCH (n) RETURN labels(n)[0] AS l, count(*) AS c")}
    rels = run_cypher("MATCH ()-[r]->() RETURN count(r) AS c")[0]["c"]
    return {
        "candidates": labels.get("Candidate", 0), "questions": labels.get("Question", 0),
        "answers": labels.get("Answer", 0), "skills": labels.get("Skill", 0), "roles": labels.get("Role", 0),
        "companies": labels.get("Company", 0), "technologies": labels.get("Technology", 0),
        "rounds": labels.get("InterviewRound", 0), "relationships": rels,
        "mongo_interviews": get_collection("raw_interviews").count_documents({}),
        "emotion_logs": get_collection("emotion_logs").count_documents({}),
    }


def graph_is_empty() -> bool:
    return run_cypher("MATCH (c:Candidate) RETURN count(c) AS n")[0]["n"] == 0


def seed_all(path: str | None = None) -> dict:
    p = Path(path) if path else find_dataset()
    data = transform(read_rows(p))
    init_neo4j_schema()
    init_mongo_indexes()
    load_neo4j(data)
    load_mongo(data)
    logger.info("Seed complete from %s", p.name)
    return {"dataset": p.name, "report": data["report"], "stats": stats(),
            "note": "Synthetic dataset: candidate/question pairs are semantically matched, not real interviews."}
```

```python
"""FastAPI application entry point."""

import logging
import threading
import time
from contextlib import asynccontextmanager

from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse
from neo4j.exceptions import ServiceUnavailable
from pymongo.errors import PyMongoError

from app.config import get_settings
from app.database.mongodb import close_mongo_client, verify_mongodb_connection
from app.database.neo4j import close_neo4j_driver, verify_neo4j_connection
from app.database.schema import init_mongo_indexes, init_neo4j_schema
from app.routes import analytics, candidates, catalog, dashboard, evaluation, graph, interviews, questions
from app.routes import database as database_routes
from app.routes import recommendations, seed
from app.utils.errors import BadRequestError, ConflictError, NotFoundError

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(name)s: %(message)s")
logger = logging.getLogger(__name__)


def _wait_for_databases(retries: int = 30, delay: float = 2.0) -> tuple[bool, bool]:
    n = m = False
    for _ in range(retries):
        n, m = verify_neo4j_connection(), verify_mongodb_connection()
        if n and m:
            break
        time.sleep(delay)
    return n, m


def _auto_seed() -> None:
    from app.services import seed_service
    try:
        if seed_service.graph_is_empty():
            logger.info("Graph is empty - seeding from dataset")
            seed_service.seed_all()
        else:
            logger.info("Graph already populated - skipping auto-seed")
    except Exception as exc:
        logger.error("Auto-seed failed: %s", exc)


@asynccontextmanager
async def lifespan(app: FastAPI):
    settings = get_settings()
    neo4j_ok, mongo_ok = _wait_for_databases()
    logger.info("Neo4j connected: %s | MongoDB connected: %s", neo4j_ok, mongo_ok)
    if neo4j_ok:
        init_neo4j_schema()
    if mongo_ok:
        init_mongo_indexes()
    if settings.auto_seed and neo4j_ok and mongo_ok:
        threading.Thread(target=_auto_seed, daemon=True).start()
    yield
    close_neo4j_driver()
    close_mongo_client()


def create_app() -> FastAPI:
    settings = get_settings()
    app = FastAPI(
        title="Interview Knowledge Graph API",
        description="Neo4j knowledge graph + MongoDB documents for synthetic interview analysis.",
        version="1.0.0", lifespan=lifespan, docs_url="/docs", redoc_url="/redoc",
    )
    app.add_middleware(CORSMiddleware, allow_origins=settings.cors_origins_list, allow_credentials=True,
                       allow_methods=["*"], allow_headers=["*"])

    @app.exception_handler(NotFoundError)
    async def _nf(_: Request, exc: NotFoundError):
        return JSONResponse(status_code=404, content={"detail": str(exc)})

    @app.exception_handler(ConflictError)
    async def _cf(_: Request, exc: ConflictError):
        return JSONResponse(status_code=409, content={"detail": str(exc)})

    @app.exception_handler(BadRequestError)
    async def _br(_: Request, exc: BadRequestError):
        return JSONResponse(status_code=400, content={"detail": str(exc)})

    @app.exception_handler(ServiceUnavailable)

                "api_prefix": settings.api_prefix}

    for module in (dashboard, candidates, questions, interviews, evaluation, graph, analytics,
```

```python
"""End-to-end validation against a running backend. Usage: python validate_system.py [base_url]"""

import json
import sys
import urllib.request

BASE = (sys.argv[1] if len(sys.argv) > 1 else "http://localhost:8000").rstrip("/")


def get(path: str):
    with urllib.request.urlopen(BASE + path, timeout=60) as r:
        return json.loads(r.read())


def main() -> int:
    failures = 0

    def check(name, fn):
        nonlocal failures
        try:
            ok, detail = fn()
        except Exception as exc:
            ok, detail = False, str(exc)[:150]
        failures += 0 if ok else 1
        print(f"[{'PASS' if ok else 'FAIL'}] {name} {detail}")

    check("health", lambda: ((h := get("/health"))["neo4j"] and h["mongodb"], str(get("/health"))))
    summary = get("/api/dashboard/summary")
    c = summary["counts"]
    check("dashboard counts > 0", lambda: (all(c[k] > 0 for k in ("candidates", "questions", "answers", "skills", "roles", "interviews", "emotion_logs")), str(c)))
    cands = get("/api/candidates")["items"]
    cid = cands[0]["candidate_id"]
    qs = get("/api/questions?page_size=5")
    for name, path in [
        ("candidate profile", f"/api/candidates/{cid}"), ("evaluation", f"/api/evaluation/{cid}"),
        ("recommendations", f"/api/recommendations/candidate/{cid}"), ("similar", f"/api/recommendations/similar/{cid}"),

        ("interviews", "/api/interviews"), ("graph", "/api/graph"), ("graph candidate", f"/api/graph/candidate/{cid}"),
        ("analytics overview", "/api/analytics/overview"), ("analytics skills", "/api/analytics/skills"),
```

```cypher
// =============================================================================
// Neo4j query pack (run in Neo4j Browser, http://localhost:7474). Each is a DA2 screenshot candidate.
// =============================================================================

// Q1 Graph overview: candidates, rounds and questions
MATCH (c:Candidate)-[:PARTICIPATES_IN]->(r:InterviewRound)-[:INCLUDES]->(q:Question)
RETURN c, r, q LIMIT 40;

// Q2 Node counts by label
MATCH (n) RETURN labels(n)[0] AS label, count(*) AS nodes ORDER BY nodes DESC;

// Q3 Relationship counts by type
MATCH ()-[r]->() RETURN type(r) AS relationship, count(*) AS total ORDER BY total DESC;

// Q4 Strongest candidates for a skill (change the skill name)
MATCH (c:Candidate)-[:ANSWERED]->(a:Answer)-[:DEMONSTRATES]->(s:Skill)
WHERE toLower(s.name) = toLower('Python')
WITH c, avg(a.score) AS avg_score, count(a) AS evidence
RETURN c.candidate_id, c.name, round(avg_score, 2) AS avg_score, evidence
ORDER BY avg_score DESC, evidence DESC LIMIT 10;

// Q5 Most difficult questions (lowest average answer score)
MATCH (a:Answer)-[:IN_RESPONSE_TO]->(q:Question)
WITH q, avg(a.score) AS avg_score, count(a) AS answers
RETURN q.question_id, q.difficulty, left(q.text, 80) AS question, round(avg_score, 2) AS avg_score
ORDER BY avg_score ASC LIMIT 10;

// Q6 Skill co-occurrence (skills tested together by the same question)
MATCH (q:Question)-[:TESTS_SKILL]->(a:Skill), (q)-[:TESTS_SKILL]->(b:Skill)
WHERE a.skill_id < b.skill_id
RETURN a.name AS skill_a, b.name AS skill_b, count(DISTINCT q) AS together
ORDER BY together DESC LIMIT 15;

// Q7 Skills required for each role (weighted)
MATCH (s:Skill)-[rel:REQUIRED_FOR]->(r:Role)
WITH r, s, rel ORDER BY rel.weight DESC
RETURN r.title AS role, collect(s.name + ' (' + rel.weight + ')') AS required_skills
ORDER BY role;

// Q8 Skill gap for the first candidate against their interviewed role(s)
MATCH (c:Candidate) WITH c ORDER BY c.candidate_id LIMIT 1
MATCH (c)-[:INTERVIEWED_FOR]->(r:Role)
MATCH (s:Skill)-[:REQUIRED_FOR]->(r)
WITH c, r, collect(DISTINCT s) AS required
WITH c, r, required, [s IN required WHERE (c)-[:HAS_SKILL]->(s) | s.name] AS matched,
     [s IN required WHERE NOT (c)-[:HAS_SKILL]->(s) | s.name] AS missing
RETURN c.candidate_id, r.title AS role, matched, missing,
       round(100.0 * size(matched) / size(required), 1) AS coverage_pct;

// Q9 Important skills (most connected: candidates + questions + roles)
MATCH (s:Skill)
OPTIONAL MATCH (c:Candidate)-[:HAS_SKILL]->(s)
WITH s, count(DISTINCT c) AS candidates
OPTIONAL MATCH (q:Question)-[:TESTS_SKILL]->(s)
WITH s, candidates, count(DISTINCT q) AS questions
OPTIONAL MATCH (s)-[:REQUIRED_FOR]->(r:Role)
RETURN s.name, candidates, questions, count(DISTINCT r) AS roles,
       candidates + questions + count(DISTINCT r) AS importance
ORDER BY importance DESC LIMIT 10;

// Q10 Candidate similarity by shared skills (Jaccard)
MATCH (a:Candidate)-[:HAS_SKILL]->(s:Skill)<-[:HAS_SKILL]-(b:Candidate)
WHERE a.candidate_id < b.candidate_id
WITH a, b, collect(s.name) AS shared, count(s) AS inter
OPTIONAL MATCH (a)-[:HAS_SKILL]->(sa:Skill) WITH a, b, shared, inter, count(sa) AS na
OPTIONAL MATCH (b)-[:HAS_SKILL]->(sb:Skill)
WITH a, b, shared, inter, na, count(sb) AS nb
RETURN a.candidate_id, b.candidate_id, shared, round(100.0 * inter / (na + nb - inter), 1) AS similarity_pct
ORDER BY similarity_pct DESC LIMIT 10;

// Q11 Role-skill analysis (required skills, questions, mean score)
MATCH (r:Role)
OPTIONAL MATCH (s:Skill)-[:REQUIRED_FOR]->(r)
WITH r, count(DISTINCT s) AS required_skills
OPTIONAL MATCH (a:Answer)-[:IN_RESPONSE_TO]->(:Question)-[:FOR_ROLE]->(r)
RETURN r.title, required_skills, count(a) AS answers, round(avg(a.score), 2) AS avg_score ORDER BY avg_score DESC;

// Q12 Company-question analysis (returns rows only if the dataset has a company column)
MATCH (c:Candidate)-[:APPLIED_TO]->(co:Company)
MATCH (c)-[:ANSWERED]->(:Answer)-[:IN_RESPONSE_TO]->(q:Question)
RETURN co.name, q.category, count(*) AS asked ORDER BY asked DESC LIMIT 20;

// Q13 Full candidate evaluation inputs (first candidate)
MATCH (c:Candidate) WITH c ORDER BY c.candidate_id LIMIT 1
MATCH (c)-[:ANSWERED]->(a:Answer)-[:IN_RESPONSE_TO]->(q:Question)
OPTIONAL MATCH (a)-[:DEMONSTRATES]->(s:Skill)
RETURN c.name, q.category, q.difficulty, a.score, collect(DISTINCT s.name) AS skills ORDER BY q.difficulty;

// Q14 Interview performance per round
MATCH (c:Candidate)-[:PARTICIPATES_IN]->(r:InterviewRound)
MATCH (a:Answer {round_id: r.round_id})
RETURN c.candidate_id, r.round_label, r.date, count(a) AS answers, round(avg(a.score), 2) AS avg_score
ORDER BY avg_score DESC LIMIT 15;

// Q15 Skill coverage per candidate for the interviewed role
MATCH (c:Candidate)-[:INTERVIEWED_FOR]->(r:Role)<-[:REQUIRED_FOR]-(s:Skill)
WITH c, r, collect(s) AS required
WITH c, r, size(required) AS total, size([s IN required WHERE (c)-[:HAS_SKILL]->(s)]) AS matched
RETURN c.candidate_id, r.title, matched, total, round(100.0 * matched / total, 1) AS coverage_pct
ORDER BY coverage_pct DESC LIMIT 15;

// Q16 Emotion vs score
MATCH (c:Candidate)-[:EXPRESSED]->(e:Emotion)
MATCH (a:Answer {answer_id: e.answer_id})
RETURN e.label, count(*) AS answers, round(avg(a.score), 2) AS avg_score, round(avg(e.confidence), 2) AS avg_confidence
ORDER BY avg_score DESC;

// Q17 UNWIND the skills property of every candidate and count holders
MATCH (c:Candidate) UNWIND c.skills AS skill
RETURN skill, count(*) AS holders ORDER BY holders DESC LIMIT 15;

// ------------------------------- GDS (Graph Data Science) ---------------------
// Q18 PageRank of skills
CALL gds.graph.project('qRank', ['Skill','Question','Candidate'],
  {TESTS_SKILL:{orientation:'NATURAL'}, HAS_SKILL:{orientation:'NATURAL'}});
CALL gds.pageRank.stream('qRank') YIELD nodeId, score
WITH gds.util.asNode(nodeId) AS n, score WHERE n:Skill
RETURN n.name AS skill, round(score, 4) AS pagerank ORDER BY pagerank DESC LIMIT 10;
CALL gds.graph.drop('qRank');

// Q19 Louvain communities of skills
CALL gds.graph.project('qComm', ['Skill','Question'], {TESTS_SKILL:{orientation:'UNDIRECTED'}});
CALL gds.louvain.stream('qComm') YIELD nodeId, communityId
WITH gds.util.asNode(nodeId) AS n, communityId WHERE n:Skill
RETURN communityId, collect(n.name) AS skills ORDER BY size(skills) DESC;
CALL gds.graph.drop('qComm');

// Q20 Node Similarity between candidates
CALL gds.graph.project('qSim', ['Candidate','Skill'], {HAS_SKILL:{orientation:'NATURAL'}});
CALL gds.nodeSimilarity.stream('qSim') YIELD node1, node2, similarity
WITH gds.util.asNode(node1) AS a, gds.util.asNode(node2) AS b, similarity WHERE a.candidate_id < b.candidate_id
RETURN a.candidate_id, b.candidate_id, round(similarity, 3) AS similarity ORDER BY similarity DESC LIMIT 10;
CALL gds.graph.drop('qSim');
```

```javascript
import { useCallback, useEffect, useState } from 'react';
import { errMsg } from '../services/api';

export function useApi(fetcher, deps = []) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [tick, setTick] = useState(0);

  useEffect(() => {
    let alive = true;
    setLoading(true);
    setError(null);
    Promise.resolve()
      .then(fetcher)
      .then((d) => alive && setData(d))
      .catch((e) => alive && setError(errMsg(e)))
      .finally(() => alive && setLoading(false));
    return () => {
      alive = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, tick]);

  const reload = useCallback(() => setTick((t) => t + 1), []);
  return { data, loading, error, reload };
}
```

```jsx
    return { nodes, links };

  const q = search.trim().toLowerCase();
  const matches = (n) => q && n.name.toLowerCase().includes(q);

  return (
    <div ref={wrap} className="card overflow-hidden bg-white">
      <ForceGraph2D
        ref={fg}
        width={width}
        height={height}
        graphData={graph}
        nodeLabel={(n) => `${n.type}: ${n.name}`}
        linkLabel={(l) => l.type}
        linkColor={() => 'rgba(100,116,139,0.35)'}
        linkDirectionalArrowLength={3}
        linkDirectionalArrowRelPos={1}
        cooldownTicks={120}
        onEngineStop={() => {
          if (!fitted.current && fg.current) {
            fg.current.zoomToFit(400, 40);
            fitted.current = true;
          }
        }}
        onNodeClick={(n) => onSelect(n)}
        onBackgroundClick={() => onSelect(null)}
        nodeCanvasObject={(node, ctx, scale) => {
          const r = RADIUS[node.type] || 4;
          const dim = q && !matches(node);
          ctx.globalAlpha = dim ? 0.12 : 1;
          ctx.beginPath();
          ctx.arc(node.x, node.y, r, 0, 2 * Math.PI);
          ctx.fillStyle = NODE_COLORS[node.type] || '#64748b';
          ctx.fill();
          if (node.id === selectedId || matches(node)) {
            ctx.lineWidth = 2 / scale;
            ctx.strokeStyle = '#0f172a';
            ctx.stroke();
          }
          const showLabel = !dim && (scale > 2.2 || ['Candidate', 'Role', 'Company'].includes(node.type) || matches(node) || node.id === selectedId);
          if (showLabel) {
            const label = node.name.length > 28 ? `${node.name.slice(0, 27)}…` : node.name;
            ctx.font = `${11 / scale}px Sans-Serif`;
            ctx.textAlign = 'center';
            ctx.textBaseline = 'top';
            ctx.fillStyle = '#1e293b';
            ctx.fillText(label, node.x, node.y + r + 1);
          }
          ctx.globalAlpha = 1;
        }}
        nodePointerAreaPaint={(node, color, ctx) => {
          ctx.fillStyle = color;
          ctx.beginPath();
          ctx.arc(node.x, node.y, (RADIUS[node.type] || 4) + 2, 0, 2 * Math.PI);
          ctx.fill();
        }}
      />
    </div>
  );
}
```

```jsx
function CandidateForm({ initial, onSaved, onClose }) {
  const editing = !!initial?.candidate_id;
  const [form, setForm] = useState(editing ? { ...initial, skills: (initial.skills || []).join(', '), experience_years: initial.experience_years ?? '' } : EMPTY);
  const [err, setErr] = useState('');
  const [busy, setBusy] = useState(false);
  const set = (k) => (e) => setForm({ ...form, [k]: e.target.value });

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true);
    setErr('');
    const body = {
      name: form.name, education: form.education || null, domain: form.domain || null, job_type: form.job_type || null,
      experience_years: form.experience_years === '' ? null : Number(form.experience_years),
      skills: form.skills.split(',').map((s) => s.trim()).filter(Boolean),
    };
    try {
      if (editing) await api.updateCandidate(initial.candidate_id, body);
      else await api.createCandidate(body);
      onSaved();
    } catch (ex) {
      setErr(errMsg(ex));
    } finally {
      setBusy(false);
    }
  };

  return (
    <form onSubmit={submit}>
      <Field label="Name"><input required className="input" value={form.name} onChange={set('name')} /></Field>
      <div className="grid grid-cols-2 gap-3">
        <Field label="Domain"><input className="input" value={form.domain || ''} onChange={set('domain')} /></Field>
```

```jsx
function InterviewForm({ initial, onSaved, onClose }) {
  const editing = !!initial;
  const cands = useApi(() => api.candidates(), []);
  const qs = useApi(() => api.questions({ page_size: 200 }), []);
  const [form, setForm] = useState({ candidate_id: '', round_label: '', target_role: initial?.target_role || '', notes: '' });
  const [qa, setQa] = useState([{ question_id: '', answer_text: '', score: '' }]);
  const [status, setStatus] = useState(initial?.status || 'completed');
  const [err, setErr] = useState('');
  const [busy, setBusy] = useState(false);

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true);
    setErr('');
    try {
      if (editing) await api.updateInterview(initial.interview_id, { target_role: form.target_role || null, notes: form.notes || null, status });
      else {
        await api.createInterview({
          candidate_id: form.candidate_id, round_label: form.round_label || null, target_role: form.target_role || null, notes: form.notes || null,
          qa: qa.filter((x) => x.question_id && x.answer_text).map((x) => ({ question_id: x.question_id, answer_text: x.answer_text, score: x.score === '' ? null : Number(x.score) })),
        });
      }
      onSaved();
    } catch (ex) {
      setErr(errMsg(ex));
    } finally {
      setBusy(false);
    }
  };

  if (!editing && (cands.loading || qs.loading)) return <Loading />;
  return (
    <form onSubmit={submit}>
      {!editing && (
        <div className="grid grid-cols-2 gap-3">

              <select className="input mb-2" value={x.question_id} onChange={(e) => setQa(qa.map((y, j) => (j === i ? { ...y, question_id: e.target.value } : y)))}>
```
